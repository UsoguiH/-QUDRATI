الأسس في القدرات وجدول الأسس المشهورة

جدول الأسس المشهورة في القدرات: احفظ قوى ٢ و ٣ و ٤ و ٥ الصغيرة. مثال: ٦٤ = ٢^٦ = ٤^٣ = ٨^٢ ، و ٨١ = ٣^٤ = ٩^٢

في هذه الصفحة ١٨ قانوناً من درس «الأسس» في قدراتي، كل قانون تحت عنوانه بصيغته ومتى تستخدمه، ١٢ منها بالشرح والمثال، ثم ٧ أسئلة أصلية بأسلوب الاختبار مع الحل خطوة بخطوة.

هدف الدرس: أن يتقن الطالب قوانين الأسس (ضرباً وقسمة وجمعاً وطرحاً)، ويحل المعادلات الأسية بطرقها الأربع، ويتعامل مع الأس السالب والزوجي والفردي، ويقارن بين أسس كبيرة بتوحيد الأساسات.

بالكلمات: احفظ قوى ٢ و ٣ و ٤ و ٥ الصغيرة

قبل كل شيء احفظ القوى الصغيرة. لو رأيت ٦٤ عرفت فوراً أنها ٢ مرفوعاً للأس ٦.

مثال: ٦٤ = ٢^٦ = ٤^٣ = ٨^٢ ، و ٨١ = ٣^٤ = ٩^٢

متى تستخدمه: أساس كل أسئلة الباب

تذكّر: العدد نفسه يُكتب بأكثر من أساس

الفخ: إذا كان ٣^س = ٨١ فما قيمة س ؟ الجواب ٤ وليس ٨١: ٨١ = ٣^٤ ، والأساس واحد ← س = ٤ ، وليست ٨١ نفسها

تساوي الأساسات

القانون: إذا كان الأساس = الأساس فإن الأس = الأس: ٢^س = ٢^٥ ⇐ س = ٥

بالكلمات: إذا تساوى الأساسان فالأسّان متساويان

إذا تشابه الأساسان على طرفي المعادلة، فالأسّان متساويان. اشطب الأساس، وساوِ بين الأسّين.

مثال: ٢^س = ٦٤ ← اكتب ٦٤ = ٢^٦ ، فتصير ٢^س = ٢^٦ ، إذن س = ٦

متى تستخدمه: معادلة أسية بأساسين متطابقين

تذكّر: حوّل العدد إلى قوة قبل أن تساوي

أسّان متساويان وأساسان مختلفان

القانون: إذا كان الأس = الأس و الأساس ≠ الأساس فإن الأس = صفر: ٣^(س+٢) = ٥^(س+٢) ⇐ س+٢ = صفر ⇐ س = −٢

بالكلمات: أساسان مختلفان وأسّ واحد ← الأس يساوي صفراً

الأسّان متشابهان والأساسان مختلفان؟ لا يتساويان إلا إذا كان الأس صفراً. ساوِ الأس بالصفر واحسب.

مثال: ٧^(س+٣) = ٤^(س+٣) ← ٧ لا تساوي ٤ ، فالأس صفر: س + ٣ = صفر ، إذن س = −٣

متى تستخدمه: معادلات بأساسين مختلفين

تذكّر: أي عدد مرفوع للصفر يساوي ١

ضرب الأساسات المتشابهة

القانون: أ^م × أ^ن = أ^(م+ن) مثال: ٤^٣ × ٤^٥ = ٤^٨ ، ٨١ × ٣^٥ = ٣^٤ × ٣^٥ = ٣^٩

بالكلمات: عند الضرب: الأساس كما هو، والأسّان يُجمعان

الأساس واحد وتضرب؟ اجمع الأسّين، والأساس يبقى كما هو. لا تضربه في نفسه.

مثال: ٣^٤ × ٣^٥ = ٣^٩ ، والأساس بقي ٣ ولم يصبح ٩

متى تستخدمه: عند الضرب نجمع الأسس

تذكّر: الضرب يجمع الأسس، لا الأساسات

قسمة الأساسات المتشابهة

القانون: أ^م ÷ أ^ن = أ^(م−ن) مثال: ٤^٧ ÷ ٤^٥ = ٤^٢ = ١٦ ، ٢^١٠ ÷ ٢^٥ = ٢^٥

بالكلمات: عند القسمة: الأساس كما هو، والأسّان يُطرحان

القسمة عكس الضرب: اطرح أس المقسوم عليه من أس المقسوم، والأساس يبقى كما هو.

مثال: ٣^٤ × ٣^٥ ÷ ٣^٧ ← ٤ + ٥ − ٧ = ٢ ، والناتج ٣^٢ = ٩

متى تستخدمه: عند القسمة نطرح الأسس

تذكّر: اجمع أسس الضرب، ثم اطرح أس القسمة

الأس السالب

القانون: أ^−ن = ١ ÷ أ^ن و (أ على ب)^−ن = (ب على أ)^ن مثال: ٢^−٥ = ١ على ٣٢ ، (٣ على ٥)^−٢ = (٥ على ٣)^٢ = ٢٥ على ٩

بالكلمات: الأس السالب ← اقلب الكسر واجعل الأس موجباً

السالب في الأس لا يعني ناتجاً سالباً. معناه اقلب: الكسر ينقلب، والأس يصير موجباً.

مثال: (٢ على ٥)^−٣ ← اقلب الكسر: (٥ على ٢)^٣ = ١٢٥ على ٨ ، ولا تضع إشارة سالبة

متى تستخدمه: لا بد من تحويل الأس السالب إلى موجب

تذكّر: الأس السالب يقلب، ولا يُسالب الناتج

الفخ: ما قيمة ٢^−٣ ؟ الجواب ١ على ٨ وليس −٨: الأس السالب يقلب: ١ ÷ ٢^٣ = ١ على ٨ ، وليست −٨

قانون التكرارات في الأسس

القانون: أ^ن + أ^ن + … + أ^ن (ك مرة) = أ^ن × ك مثال: ٤^٦+٤^٦+٤^٦+٤^٦ = ٤^٦×٤ = ٤^٧

بالكلمات: المجموع = القوة المكررة × عدد المرات

القوة نفسها مكررة بالجمع؟ لا تلمس الأسس. عُدّ كم مرة تكررت، واضرب فيها.

مثال: ٥^٧ + ٥^٧ + ٥^٧ + ٥^٧ + ٥^٧ = ٥^٧ × ٥ = ٥^٨

متى تستخدمه: الجمع لا يعني جمع الأسس

تذكّر: الجمع لا يجمع الأسس أبداً

جمع وطرح الأساسات المتشابهة

القانون: نأخذ العامل المشترك (أصغر أس) أو نحسب كل قيمة على حدة: ٣^٦+٣^٤ = ٣^٤(٣^٢+١) = ٨١×١٠ = ٨١٠ ، ٢^٥+٢^٣ = ٣٢+٨ = ٤٠

بالكلمات: أخرج أصغر قوة عاملاً مشتركاً، أو احسب كل قوة وحدها

الأسّان مختلفان؟ أخرج أصغر قوة عاملاً مشتركاً، أو احسب كل قوة على حدة ثم اجمع.

مثال: ٢^٩ + ٢^٧ = ٢^٧ × (٢^٢ + ١) = ١٢٨ × ٥ = ٦٤٠

متى تستخدمه: جمع أو طرح قوى نفس الأساس

تذكّر: ٢^٩ + ٢^٧ ليست ٢^١٦

جرّب بنفسك: ما قيمة ٤^٦ + ٤^٦ + ٤^٦ + ٤^٦ ؟

الجواب

٤^٧

تكررت أربع مرات: ٤^٦ × ٤ = ٤^٧ ، والأسس لا تُجمع في الجمع

الأس الزوجي والفردي

القانون: الأس الزوجي للعدد السالب يعطي ناتجاً موجباً: (−٢)^٤ = ١٦ ؛ الأس الفردي يعطي ناتجاً سالباً: (−٢)^٣ = −٨

بالكلمات: أس زوجي ← موجب ، وأس فردي ← سالب

السالب داخل قوس؟ الأس الزوجي يعطي ناتجاً موجباً، والأس الفردي يعطي ناتجاً سالباً. انظر للأس فقط.

مثال: (−٣)^٨ موجبة لأن ٨ زوجي، و (−٣)^٧ سالبة لأن ٧ فردي ← الأولى أكبر

متى تستخدمه: إشارات القوى

تذكّر: القوس يضم السالب، وبلا قوس الإشارة خارج

العدد المرفوع لأسّ الأسّ

القانون: أ^ب^ج نحسب أس الأساس أولاً: ٢^٢^٣ ⇒ أولاً ٢^٣=٨ ثم ٢^٨=٢٥٦

بالكلمات: ابدأ بالأس الأعلى، ثم ارفع الأساس إلى الناتج

لا يوجد قوس؟ ابدأ من الأعلى: احسب الأس أولاً، ثم ارفع الأساس إلى الناتج.

مثال: ٢^٣^٢ ← ٣^٢ = ٩ أولاً، ثم ٢^٩ = ٥١٢ ، لا ٦٤

متى تستخدمه: البرج الأسي بدون أقواس

تذكّر: بالقوس تضرب الأسس، وبدونه تبدأ بالأعلى

الفخ: ما قيمة ٢^٢^٣ ؟ الجواب ٢٥٦ وليس ٦٤: ٢^٣ = ٨ أولاً، ثم ٢^٨ = ٢٥٦ ، و ٦٤ ناتج القوس

العدد المرفوع لأكثر من أس

القانون: (أ^م)^ن = أ^(م×ن) مثال: (٢^٣)^٤ = ٢^١٢

بالكلمات: قوس عليه أس ← اضرب الأسّين

قوس عليه أس؟ اضرب الأسّين. بهذه الأداة تردّ أي أساس إلى أساس أصغر.

مثال: ٨^١٥ ← ٨ = ٢^٣ ، إذن (٢^٣)^١٥ = ٢^٤٥

متى تستخدمه: تبسيط الأقواس الأسية

تذكّر: الأسس تُضرب هنا، ولا تُجمع

المقارنة بين الأسس الكبيرة

القانون: ثلاث طرق: ١) تصغير الأسس بقسمتها على أكبر قاسم ٢) جعل الأساسات متساوية ٣) حذف المتشابهات من طرفي المقارنة

بالكلمات: وحّد الأساس ← قارن الأسّين فقط

لا تحسب قيمة ضخمة أبداً. ردّ الطرفين إلى أساس واحد، ثم صاحب الأس الأكبر هو الأكبر.

مثال: ٨^١٥ = ٢^٤٥ و ٤^٢٢ = ٢^٤٤ ← الأساس واحد، فالأولى أكبر

متى تستخدمه: بند «قارن بين» في الأسس

تذكّر: الأساس الأصغر لا يعني قيمة أصغر

الفخ: أيهما أكبر: ٤^٥ أم ٢^١١ ؟ الجواب ٢^١١ وليس ٤^٥: ٤^٥ = (٢^٢)^٥ = ٢^١٠ ، والأساس واحد ← ٢^١١ أكبر

بقية قوانين الأسس

٦ قوانين أخرى في درس «الأسس»، كلٌّ بصيغته ومتى تستخدمه.

الأس صفر

القانون: أي عدد^صفر = ١ مثال: ٥^س = ١ ⇐ س = صفر

متى تستخدمه: حل المعادلات الأسية

تساوي الأسس

القانون: إذا كان الأس = الأس فإن الأساس = الأساس: ٣^٧ = س^٧ ⇐ س=٣ (أس فردي: حل واحد) ، ٣^٦ = س^٦ ⇐ س = ±٣ (أس زوجي: حلّان)

متى تستخدمه: معادلة أسية بأسّين متطابقين

التخمين

القانون: نخمّن قيمة س التي تحقق المعادلة: ٣^س = ٨١ ⇐ س = ٤

متى تستخدمه: إذا فشلت الطرق السابقة

ضرب الأساسات المختلفة

القانون: يمكن ضربها إذا تساوت الأسس: ٢^٥ × ٣^٥ = ٦^٥

متى تستخدمه: توحيد الأساس بالضرب

توزيع الأس

القانون: (س^أ × ص^ب × ع^ج)^ن = س^أن × ص^بن × ع^جن مثال: (س^٣ص^٤ع^٥)^٢ = س^٦ص^٨ع^١٠

متى تستخدمه: فك الأقواس

حركات في القدرات

القانون: ٩^٢=٨١ ، ٩٩^٢=٩٨٠١ ، ٩٩٩^٢=٩٩٨٠٠١ ؛ مجموع أرقام الناتج = ٩ × عدد مرات التكرار

متى تستخدمه: مربع الأعداد المكوّنة من تسعات

حيل تختصر حل أسئلة الأسس

  • في المقارنة بين أسس ضخمة لا تحسب شيئاً: وحّد الأساس (٨^١٥=٢^٤٥ و ٤^٢٢=٢^٤٤) أو صغّر الأسس بقسمتها على أكبر قاسم مشترك، أو احذف المتشابهات من الطرفين.
  • الضرب ⇐ نجمع الأسس ، أما الجمع ⇐ قانون التكرارات: أ^ن مكرراً ك مرة = ك × أ^ن — أشهر خطأ في الاختبار.
  • انتبه لموضع السالب: (−٥)^٦ موجب أما −٥^٦ فسالب (السالب خارج الأس).
  • ٢^٣^٢ ≠ (٢^٣)^٢ — في البرج الأسي ابدأ من الأعلى.
  • الأس السالب يعني مقلوب وليس إشارة سالبة: ٢^−٣ = ١ على ٨ وليس −٨.

أسئلة الأسس محلولة

٧ أسئلة أصلية من بنك قدراتي، بأسلوب الاختبار. حلّها أولاً، ثم افتح الحل.

السؤال ١: إذا كان ٢^س = ٦٤ فما قيمة س ؟

اعرض الحل

الإجابة الصحيحة: (ب) ٦

نكتب ٦٤ كقوة للعدد ٢ : ٦٤ = ٢^٦

إذن ٢^س = ٢^٦

الأساس = الأساس ⇐ الأس = الأس ⇐ س = ٦

السؤال ٢: ما قيمة ٣^٤ × ٣^٥ ÷ ٣^٧ ؟

اعرض الحل

الإجابة الصحيحة: (ب) ٩

عند الضرب نجمع الأسس وعند القسمة نطرحها:

٣^(٤+٥−٧) = ٣^٢ = ٩

السؤال ٣: إذا كان ٧^(س+٣) = ٤^(س+٣) فما قيمة س ؟

اعرض الحل

الإجابة الصحيحة: (جـ) −٣

الأس = الأس ولكن الأساس ≠ الأساس (٧ ≠ ٤)

إذن لا بد أن يكون الأس = صفر

س + ٣ = صفر ⇐ س = −٣

السؤال ٤: ما قيمة ٢^٩ + ٢^٧ ؟

نأخذ العامل المشترك وهو أصغر أس ٢^٧ :

٢^٧(٢^٢ + ١) = ١٢٨ × ٥ = ٦٤٠

السؤال ٥: قارن بين ٨^١٥ و٤^٢٢

قارن بين القيمتين:

القيمة الأولى: ٨^١٥
القيمة الثانية: ٤^٢٢
اعرض الحل

القيمة الأولى = ٨^١٥ = (٢^٣)^١٥ = ٢^٤٥

القيمة الثانية = ٤^٢٢ = (٢^٢)^٢٢ = ٢^٤٤

الأساس واحد ⇐ صاحب الأس الأكبر أكبر ⇐ القيمة الأولى أكبر

السؤال ٦: قارن بين ٣٦^٣٠ و٦^٦٠

قارن بين القيمتين:

القيمة الأولى: ٣٦^٣٠
القيمة الثانية: ٦^٦٠
اعرض الحل

الإجابة الصحيحة: (جـ) القيمتان متساويتان

٣٦ = ٦^٢

القيمة الأولى = (٦^٢)^٣٠ = ٦^(٢×٣٠) = ٦^٦٠

وهي نفس القيمة الثانية ⇐ القيمتان متساويتان

السؤال ٧: ما قيمة ٢^٣^٢ ؟

اعرض الحل

الإجابة الصحيحة: (جـ) ٥١٢

هنا العدد مرفوع لأسّ الأسّ (لا توجد أقواس على ٢ و٣)، فنحسب أس الأساس أولاً:

٣^٢ = ٩

ثم ٢^٩ = ٥١٢

الخيار (أ) ٦٤ هو ناتج (٢^٣)^٢ = ٢^٦ وهو الفخ.

وجدت خطأ في سؤال أو حل؟ هكذا تبلّغ عنه وهكذا نصحّحه.

المزيد: سؤالان آخران على الأسس مع طريقة الحل، واختبار قدرات كمي تجريبي مجاني بمؤقّت.

احفظ قوى ٢ و ٣ و ٤ و ٥ الصغيرة (بالرموز: ٢^٢=٤ ، ٢^٣=٨ ، ٢^٤=١٦ ، ٢^٥=٣٢ ، ٢^٦=٦٤ ، ٢^٧=١٢٨ ، ٢^٨=٢٥٦ ، ٢^١٠=١٠٢٤ ؛ ٣^٢=٩ ، ٣^٣=٢٧ ، ٣^٤=٨١ ، ٣^٥=٢٤٣ ؛ ٤^٢=١٦ ، ٤^٣=٦٤ ؛ ٥^٢=٢٥ ، ٥^٣=١٢٥ ، ٥^٤=٦٢٥ ، ٥^٥=٣١٢٥ ؛ ٦^٢=٣٦ ، ٦^٣=٢١٦ ؛ ٧^٢=٤٩ ، ٧^٣=٣٤٣ ؛ ٨^٢=٦٤ ؛ ٩^٢=٨١ ، ٩^٣=٧٢٩). مثال: ٦٤ = ٢^٦ = ٤^٣ = ٨^٢ ، و ٨١ = ٣^٤ = ٩^٢.
ما قانون ضرب الأسس في القدرات؟
عند الضرب: الأساس كما هو، والأسّان يُجمعان (بالرموز: أ^م × أ^ن = أ^(م+ن)). مثال: ٣^٤ × ٣^٥ = ٣^٩ ، والأساس بقي ٣ ولم يصبح ٩.
كيف أتعامل مع الأس السالب في القدرات؟
الأس السالب ← اقلب الكسر واجعل الأس موجباً (بالرموز: أ^−ن = ١ ÷ أ^ن و (أ على ب)^−ن = (ب على أ)^ن). مثال: (٢ على ٥)^−٣ ← اقلب الكسر: (٥ على ٢)^٣ = ١٢٥ على ٨ ، ولا تضع إشارة سالبة.
ما الفخ في أسئلة الأسس في القدرات؟
إذا كان ٣^س = ٨١ فما قيمة س ؟ الجواب ٤ وليس ٨١: ٨١ = ٣^٤ ، والأساس واحد ← س = ٤ ، وليست ٨١ نفسها.
كيف أحل أسئلة الأسس بسرعة في القدرات؟
في المقارنة بين أسس ضخمة لا تحسب شيئاً: وحّد الأساس (٨^١٥=٢^٤٥ و ٤^٢٢=٢^٤٤) أو صغّر الأسس بقسمتها على أكبر قاسم مشترك، أو احذف المتشابهات من الطرفين.
